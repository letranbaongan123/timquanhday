async function getLocation() { const locationText = document.getElementById("location"); const groups = document.getElementById("groups"); locationText.innerHTML = "Đang lấy vị trí..."; if (!navigator.geolocation) { locationText.innerHTML = "Trình duyệt không hỗ trợ GPS."; return; } navigator.geolocation.getCurrentPosition( async function(position) { const lat = position.coords.latitude; const lng = position.coords.longitude; groups.style.display = "grid"; locationText.innerHTML = "✅ Đã xác nhận vị trí"; try { await fetch("/api/location", { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ lat, lng }) }); } catch(error) { console.log(error); } }, function() { locationText.innerHTML = "❌ Bạn cần bật vị trí để xem nhóm."; groups.style.display = "none"; } ); }
